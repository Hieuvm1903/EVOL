# Optional: enables the "Gesture capture" mode on the Photobooth tab.
# Install with: pip install -r requirements-gesture.txt
#

# mediapipe, opencv) that can be finicky on some local Windows setups.

# falling back to plain click-to-capture. Community Cloud (Linux) installs
# these cleanly, so you can also just skip local install and only enable
# gesture mode in the deployed app (add this file's contents to
# requirements.txt before deploying).

mediapipe
opencv-python-headless
